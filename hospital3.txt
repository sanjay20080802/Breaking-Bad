function addPatient(){

    let name =
    document.getElementById("name").value;

    let age =
    document.getElementById("age").value;

    let disease =
    document.getElementById("disease").value;

    if(name == "" || age == "" || disease == ""){
        alert("Fill all details");
        return;
    }

    let li =
    document.createElement("li");

    li.textContent =
    "Name: " + name +
    " | Age: " + age +
    " | Disease: " + disease;

    document
    .getElementById("patientList")
    .appendChild(li);

    document.getElementById("name").value = "";
    document.getElementById("age").value = "";
    document.getElementById("disease").value = "";
}
